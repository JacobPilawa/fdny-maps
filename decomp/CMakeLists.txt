cmake_minimum_required(VERSION 3.20)
project(fdny_training LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL REQUIRED IMPORTED_TARGET sdl2)
pkg_check_modules(JSON REQUIRED IMPORTED_TARGET json-c)
find_library(OPENGL_FRAMEWORK OpenGL REQUIRED)
find_library(CORETEXT_FRAMEWORK CoreText REQUIRED)
find_library(COREGRAPHICS_FRAMEWORK CoreGraphics REQUIRED)
add_executable(fdny-training src/main.cpp)
target_link_libraries(fdny-training PRIVATE PkgConfig::SDL PkgConfig::JSON ${OPENGL_FRAMEWORK} ${CORETEXT_FRAMEWORK} ${COREGRAPHICS_FRAMEWORK})
target_compile_options(fdny-training PRIVATE -ffp-contract=off -Wno-deprecated-declarations)
target_compile_definitions(fdny-training PRIVATE FDNY_ASSETS="${CMAKE_CURRENT_SOURCE_DIR}/assets/training")

include(CTest)
if(BUILD_TESTING)
 foreach(check actor_render_flags victim_pickup_effects victim_release_order victim_passout_order extended_zero victim_rescue_statistics)
  add_executable(${check}-check tools/${check}_check.cpp)
  target_compile_options(${check}-check PRIVATE -ffp-contract=off -UNDEBUG)
  add_test(NAME ${check} COMMAND ${check}-check)
 endforeach()
 add_test(NAME victim_release_frame COMMAND fdny-training --victim-release-frame-check)
 add_test(NAME victim_passout_frame COMMAND fdny-training --victim-passout-frame-check)
 add_test(NAME victim_frame_ops COMMAND fdny-training --victim-frame-ops-check)
endif()
